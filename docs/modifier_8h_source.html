57template <ModifierOp Op = ModifierOp::Multiplicative>
58struct ModifiableValue {
59 static constexpr double identity = (Op == ModifierOp::Additive) ? 0.0 : 1.0;
60
61 double base = 0.0;
62 double sum = identity;
63 double value = 0.0;
64
65 std::vector<std::pair<entt::entity, double>> modifiers;
66
67 void recalculate() {
68 sum = std::accumulate(modifiers.begin(), modifiers.end(), identity,
69 [](double acc, const auto& m) { return acc + m.second; });
70 if constexpr (Op == ModifierOp::Additive) {
71 value = base + sum;
72 } else {
73 // Then we can't go negative with modifiers?
74 sum = std::max(sum, 0.);
75 value = base * sum;
76 }
77 }

79 // NOLINTNEXTLINE(google-explicit-constructor)
80 operator double() const { return value; }
